BEGIN TRANSACTION;
CREATE TABLE IF NOT EXISTS "Dim_Drugs" (
	"drug_id"	INTEGER,
	"drug_name"	VARCHAR(100) NOT NULL,
	"concomitant_drugs"	VARCHAR(250),
	PRIMARY KEY("drug_id" AUTOINCREMENT)
);
CREATE TABLE IF NOT EXISTS "Dim_Events" (
	"event_id"	INTEGER,
	"adr_code"	VARCHAR(20) NOT NULL,
	"seriousness"	VARCHAR(10) CHECK("seriousness" IN ('mild', 'moderate', 'severe', 'fatal')),
	PRIMARY KEY("event_id" AUTOINCREMENT)
);
CREATE TABLE IF NOT EXISTS "Dim_Patients" (
	"patient_id"	INTEGER,
	"patient_age"	INT,
	"gender"	VARCHAR(10),
	"age_group"	VARCHAR(20),
	PRIMARY KEY("patient_id" AUTOINCREMENT)
);
CREATE TABLE IF NOT EXISTS "Fact_ADR_Reports" (
	"fact_id"	INTEGER,
	"report_id"	VARCHAR(20) NOT NULL,
	"patient_id"	INT NOT NULL,
	"drug_id"	INT NOT NULL,
	"event_id"	INT NOT NULL,
	"dosage"	VARCHAR(50),
	"duration_days"	INT CHECK("duration_days" >= 0),
	"onset_days"	INT CHECK("onset_days" >= 0),
	"report_date"	DATE,
	PRIMARY KEY("fact_id" AUTOINCREMENT),
	FOREIGN KEY("drug_id") REFERENCES "Dim_Drugs"("drug_id") ON DELETE CASCADE,
	FOREIGN KEY("event_id") REFERENCES "Dim_Events"("event_id") ON DELETE CASCADE,
	FOREIGN KEY("patient_id") REFERENCES "Dim_Patients"("patient_id") ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS "idx_dim_drug" ON "Dim_Drugs" (
	"drug_name",
	"concomitant_drugs"
);
CREATE INDEX IF NOT EXISTS "idx_dim_event" ON "Dim_Events" (
	"adr_code",
	"seriousness"
);
CREATE INDEX IF NOT EXISTS "idx_dim_patient" ON "Dim_Patients" (
	"patient_age",
	"gender"
);
COMMIT;

-- ============================================================
-- DATA TRANSFORMATIONS & CLEANING
-- ============================================================

-- 1. Standardize Drug Names (Brand to Generic)
UPDATE Dim_Drugs
SET drug_name = CASE 
    WHEN drug_name IN ('Ozempic', 'Wegovy') THEN 'Semaglutide'
    WHEN drug_name = 'Glucophage' THEN 'Metformin'
    ELSE drug_name
END;

-- 2. Populate Patient Age Bins
UPDATE Dim_Patients
SET age_group = CASE 
    WHEN patient_age < 18 THEN 'Pediatric'
    WHEN patient_age BETWEEN 18 AND 44 THEN 'Adult (18-44)'
    WHEN patient_age BETWEEN 45 AND 64 THEN 'Middle Aged (45-64)'
    ELSE 'Elderly (65+)'
END;
